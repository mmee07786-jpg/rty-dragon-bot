import discord from discord.ext import commands from fastapi import FastAPI, HTTPException from fastapi.responses import HTMLResponse import uvicorn import json import os import asyncio app = FastAPI() intents = discord.Intents.default() intents.members = True bot = commands.Bot(command_prefix="!", intents=intents) SETTINGS_FILE = "server_settings.json" def load_all_settings(): if os.path.exists(SETTINGS_FILE): with open(SETTINGS_FILE, "r", encoding="utf-8") as f: try: return json.load(f) except: return {} return {} def save_all_settings(data): with open(SETTINGS_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) # 1. واجهة الموقع الكاملة (HTML + CSS + JS) مدمجة بالبايثون مع دعم الصور والأزرار المتعددة @app.get("/", response_class=HTMLResponse) async def serve_dashboard(): return """ ItzF18 Bot - لوحة التحكم
⚡ ItzF18 Dashboard
🤖 دعوة البوت تسجيل الدخول
إعدادات السيرفر
تحكم بأنظمة البوت والصور والأزرار المتعددة للتكت والترحيب.

🌐 معرف السيرفر (Guild ID)

👋 نظام الترحيب (Welcome)

🚀 نظام البوست (Boost)

🎫 نظام التكت (Tickets)

""" # 2. استقبال الإعدادات وتحديثها @app.post("/api/update-settings") async def update_settings(payload: dict): guild_id = str(payload.get("guild_id")) new_settings = payload.get("settings", {}) if not guild_id: raise HTTPException(status_code=400, detail="Missing guild_id") all_data = load_all_settings() all_data[guild_id] = new_settings save_all_settings(all_data) return {"status": "success"} # 3. أحداث البوت (الترحيب، البوست، وتطبيق الصور) @bot.event async def on_ready(): print(f"Logged in as {bot.user.name}") @bot.event async def on_member_join(member): data = load_all_settings() guild_id = str(member.guild.id) if guild_id in data and data[guild_id].get("welcome_enabled", False): settings = data[guild_id] channel = member.guild.get_channel(int(settings.get("welcome_channel", 0))) if channel: msg = settings.get("welcome_message", "").replace("{user}", member.mention).replace("{server}", member.guild.name) img_url = settings.get("welcome_img", "") if img_url: embed = discord.Embed(description=msg, color=0x5865F2) embed.set_image(url=img_url) await channel.send(embed=embed) else: await channel.send(msg) @bot.event async def on_member_update(before: discord.Member, after: discord.Member): if before.premium_since is None and after.premium_since is not None: data = load_all_settings() guild_id = str(after.guild.id) if guild_id in data and data[guild_id].get("boost_enabled", False): settings = data[guild_id] channel = after.guild.get_channel(int(settings.get("boost_channel", 0))) if channel: msg = settings.get("boost_message", "").replace("{user}", after.mention) img_url = settings.get("boost_img", "") if img_url: embed = discord.Embed(description=msg, color=0xF47FFF) embed.set_image(url=img_url) await channel.send(embed=embed) else: await channel.send(msg) #
