import discord from discord.ext import commands from fastapi import FastAPI, HTTPException from fastapi.responses import HTMLResponse import uvicorn import json import os import asyncio # إعداد تطبيق FastAPI والبوت معاً app = FastAPI() intents = discord.Intents.default() intents.members = True bot = commands.Bot(command_prefix="!", intents=intents) SETTINGS_FILE = "server_settings.json" def load_all_settings(): if os.path.exists(SETTINGS_FILE): with open(SETTINGS_FILE, "r", encoding="utf-8") as f: try: return json.load(f) except: return {} return {} def save_all_settings(data): with open(SETTINGS_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=4) # 1. عرض صفحة الموقع مباشرة عند فتح الرابط بالمتصفح (HTML مدمج داخل البايثون) @app.get("/", response_class=HTMLResponse) async def serve_dashboard(): return """ ItzF18 Bot - لوحة التحكم المدمجة

✨ ItzF18 Bot Dashboard

لوحة تحكم مدمجة لإدارة إعدادات السيرفر وإرسالها للبوت فوراً

👋 نظام الترحيب (Welcome)

🚀 نظام البوست (Boost)

🎫 نظام التكت (Tickets)

""" # 2. استقبال الإشارة وتحديث الإعدادات @app.post("/api/update-settings") async def update_settings(payload: dict): guild_id = str(payload.get("guild_id")) new_settings = payload.get("settings", {}) if not guild_id: raise HTTPException(status_code=400, detail="Missing guild_id") all_data = load_all_settings() all_data[guild_id] = new_settings save_all_settings(all_data) return {"status": "success", "message": "All settings updated successfully!"} # 3. أحداث البوت (الترحيب والبوست) @bot.event async def on_ready(): print(f"Logged in as {bot.user.name} (ID: {bot.user.id})") @bot.event async def on_member_join(member): data = load_all_settings() guild_id = str(member.guild.id) if guild_id in data and data[guild_id].get("welcome_enabled", False): settings = data[guild_id] channel = member.guild.get_channel(int(settings.get("welcome_channel", 0))) if channel: msg = settings.get("welcome_message", "Welcome {user}!").replace("{user}", member.mention).replace("{server}", member.guild.name) await channel.send(msg) @bot.event async def on_member_update(before: discord.Member, after: discord.Member): if before.premium_since is None and after.premium_since is not None: data = load_all_settings() guild_id = str(after.guild.id) if guild_id in data and data[guild_id].get("boost_enabled", False): settings = data[guild_id] channel = after.guild.get_channel(int(settings.get("boost_channel", 0))) if channel: msg = settings.get("boost_message", "Thanks for boosting!").replace("{user}", after.mention) await channel.send(msg) # تشغيل البوت وخادم الويب معاً في نفس الوقت async def main(): # تشغيل سيرفر FastAPI بالخلفية config = uvicorn.Config(app, host="0.0.0.0", port=5000, log_level="warning") server = uvicorn.Server(config) # تشغيل الاثنين سوية await asyncio.gather( server.serve(), bot.start("TOKEN_BOT_MTAA") # استبدل هذا التوكن بتوكن بوتك الحقيقي ) if __name__ == "__main__": asyncio.run(main())
